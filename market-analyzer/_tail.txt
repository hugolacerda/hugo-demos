<script>
  function toggleTab(trigger) {
    const allTriggers = document.querySelectorAll('.tab-trigger');
    const allBodies = document.querySelectorAll('.tab-body');

    allTriggers.forEach(t => t.classList.remove('active'));
    allBodies.forEach(b => b.classList.remove('open'));

    trigger.classList.add('active');
    trigger.nextElementSibling.classList.add('open');
  }
</script>
